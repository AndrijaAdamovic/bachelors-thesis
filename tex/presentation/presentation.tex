\section{Willemsova Fundamentalna Lema}

\subsection{Uvod}
\begin{frame}
    \frametitle{Lema - Uvod}
    \begin{itemize}
        \item Willems et. al - \textit{A note on persistency of excitation}
        \item Snimljena ulazno-izlazna trajektorija LTI sustava koji je \textit{perzistentno pobuđen} može parametrizirati bilo koje moguće trajektorije te duljine
    \end{itemize}
    
    \begin{block}{Definicija - Perzistentna pobuda}
        Za niz ulaza $u(0), \dots, u(T-1) \in \mathbb{R}^{mT}$, $m$ - broj ulaza u sustav, kažemo da je perzistentna pobuda reda $k \in \{1, \dots, T\}$
        ako je Hankelova matrica:
        \[H_{k}(u_{[0, T-1]}) =
        \begin{bmatrix}
        u(0) & u(1) & \cdots & u(T-k) \\
        u(1) & u(2) & \cdots & u(T-k+1)\\
        \vdots & \vdots &  & \vdots \\
        u(k-1) & u(k) & \cdots & u(T-1)
        \end{bmatrix}\]    

        punog retčanog ranga.

\end{block}

\end{frame}

\subsection{Formulacija}

\begin{frame}
    \frametitle{Lema - Formulacija}
    \begin{itemize}
        \item onznačimo ulazno-izlaznu trajektoriju s $(u_{[0, T-1]}, y_{[0, T-1]})$ - par vektor stupaca
        \item označimo s $N\in\mathbb{N}$ neku gornju među na broj stanja sustava
        \begin{block}{Willemsova Fundamentalna Lema}
        Neka je dan kontrolabilan LTI sustav te $(u_{[0, T-1]}, y_{[0, T-1]})$ ulazno-izlazna trajektorija. Neka je $L\in\{1, ..., T\}$.
        Ako je ulaz $u_{[0, T-1]}$ perzistentna pobuda reda $N+L$ tada je $(\bar{u}_{[0, L-1]}, \bar{y}_{[0, L-1]})$ trajektorija sustava ako i samo ako:
        \[\begin{bmatrix} \bar{u}_{[0, L-1]} \\ \bar{y}_{[0, L-1]} \end{bmatrix} = \begin{bmatrix} H_{L}(u_{[0, T-1]}) \\ H_{L}(y_{[0, T-1]})\end{bmatrix}g \]
        za neki $g \in \mathbb{R}^{T-L+1}$
        \end{block}

        \item svaka trajektorija duljine $L$ se nalazi u slici Hankelove matrice tj. stupci
        Hankelove matrice razapinju prostor svih mogućih trajektorija duljine L.
        \item dokaz Leme u završnom radu
    \end{itemize}
\end{frame}

\section{Simulacija LTI sustava temeljena na podacima}
\subsection{Formulacija problema simulacije}
\begin{frame}
    \frametitle{Formulacija problema simulacije}
    \begin{itemize}
        \item poznata početna trajektorija \((\bar{u}_{[0, L_{ini}-1]}, \bar{y}_{[0, L_{ini}-1]})\), duljine $L_{ini}$. Želimo simulirati nastavak trajektorije duljine
              $L_{ref}$ ako je poznat ulaz \(\bar{u}_{[L_{ini}, L-1]}\), gdje $L = L_{ini} + L_{ref}$
        \item imamo snimljenu trajektoriju \((u_{[0, T-1]}, y_{[0, T-1]})\),  duljine $T$ čiji je ulaz perzistentna pobuda reda $N + L$
        \item particioniramo Hankelovu matricu na sljedeći način:
              \[\begin{bmatrix} H_{L}(u_{[0, T-1]}) \\ H_{L}(y_{[0, T-1]})\end{bmatrix} = \begin{bmatrix}U_p \\ U_f \\ Y_p \\ Y_f\end{bmatrix},\]
              gdje $U_p, Y_p, U_f, Y_f$ imaju redom $mL_{ini}$, $pL_{ini}, mL_{ref}$, $pL_{ref}$ redaka, $m$ - broj ulaza, a $p$ broj izlaza sustava
    \end{itemize}
\end{frame}

\subsection{Simulacija LTI sustava}
\begin{frame}
    \frametitle{Simulacija LTI sustava}
    \begin{itemize}
        \item tada postoji $g\in\mathbb{R}^{T-L+1}$ takav da :
                \[\begin{bmatrix}U_p \\ Y_p \\ U_f\end{bmatrix}g = \begin{bmatrix} \bar{u}_{[0, L_{ini} -1]} \\ \bar{y}_{[0, L_{ini} -1]} \\ \bar{u}_{[L_{ini}, L-1]}\end{bmatrix}.\]
        \item nastavak (budućnost) trajektorije je tada \(\bar{y}_{[L_{ini}, L-1]} = Y_{f}g\)

\section{Prediktivno upravljanje temeljeno podacima - DeePC}
\subsection{Uvod i usporedba s MPC}
\begin{frame}
    \frametitle{DeePC - uvod}
    \begin{itemize}
        \item vidjeli smo da je moguće simulirati buduću trajektoriju sustava temeljeno samo na podacima
        \item pitanje je može li se to koristiti za prediktivno upravljanje poput MPC-a
        \item odgovor je da i prvo su ga formulirali Coulson, et al. u članku \textit{Data-Enabled Predictive Control: In the Shallows of the DeePC}
        
        \begin{equation}
            \begin{aligned}
                \min_{g,y,u} \quad  &\sum_{j=0}^{L_{ref}-1} \left(\norm{\bar{y}_j - r_{t+j}}^{2}_{Q} + \norm{\bar{u}_j}^{2}_{R}\right) \\
                \text{subject to:} \quad & \begin{bmatrix} U_p \\ Y_p \\ U_f \\ Y_f\end{bmatrix}g = \begin{bmatrix} \bar{u}_{ini} \\ \bar{y}_{ini} \\ \bar{u} \\ \bar{y}\end{bmatrix}\\
                &\bar{y}_j \in \mathcal{Y}, \quad j=0,\dots,L_{ref}-1, \\
                &\bar{u}_j \in \mathcal{U}, \quad j=0,\dots,L_{ref}-1, \\
            \end{aligned}
            \label{eq:deepc}
        \end{equation}

        \item ovo je \textit{de facto} MPC jedino umjesto modela sustava koriste se podaci, tj. Hankelove matrice
    \end{itemize}
\end{frame}

\subsection{DeePC za LTI sustave}

\begin{frame}[t]
    \frametitle{DeePC za LTI sustave}
    \vspace{-0.6em}
    \begingroup
    \setlength{\intextsep}{0pt}
    \begin{algorithm}[H]
        \scriptsize
        \caption{Data-Enabled predictive control (DeePC)}
        \begin{algorithmic}[1]
            \Require Requirements (i) - (vii) listed above
            \For{$t = 0, 1, ...$}
                \State Solve \eqref{eq:deepc} for the vector g
                \State Compute optimal input sequence $\bar{u}^{*} = U_{f}g$
                \State Apply the first input $u(t) = \bar{u}^{*}(0)$
                \State Update the initial trajectory $(u_{ini}, y_{ini})$ with new inputs/measurements
            \EndFor
        \end{algorithmic}
    \end{algorithm}
    \endgroup
    \vspace{-0.6em}
    \begin{itemize}\small
        \item primjer za dvostruki MDS sustav (\ref{fig:dsmd}) sa jediničnim step-om kao referenca

\subsection{Regularizacije}
\begin{frame}
    \frametitle{DeePC Regularizacije}
    \begin{itemize}
        \item zbog šuma i nelinearnosti stvarnih sustava stupci Hankelove matrice ne razapinju potprostor s trajektorijama već, načelno, cijeli prostor
        \item Lema tada ne vrijedi direktno time DeePC ne funkcionira dobro, stoga se uvode regularizacije 
        \begin{equation}
            \begin{aligned}
                \min_{g,y,u, \sigma_y} \quad  &\sum_{j=0}^{L_{ref}-1} \left(\norm{\bar{y}_j - r_{t+j}}^{2}_{Q} + \norm{\bar{u}_j}^{2}_{R}\right) \\
                &+ \textcolor{red}{\lambda_g\norm{g}_2^{2}}\ + \textcolor{orange}{\lambda_y\norm{\sigma_y}_2^{2}}\\
                \text{subject to:} \quad & \begin{bmatrix} U_p \\ Y_p \\ U_f \\ Y_f\end{bmatrix}g = \begin{bmatrix} \bar{u}_{ini} \\ \bar{y}_{ini} + \sigma_y \\ \bar{u} \\ \bar{y}\end{bmatrix}\\
                &\bar{y}_j \in \mathcal{Y}, \quad j=0,\dots,L_{ref}-1; \\
                &\bar{u}_j \in \mathcal{U}, \quad j=0,\dots,L_{ref}-1; \\
            \end{aligned}
            \label{eq:deepc-reg}
        \end{equation}
    \end{itemize}
\end{frame}
